\# Команды работы с удаленным Git-репозиторием



\# Проверить удаленные подключения

git remote -v



\# Отправить новые коммиты

git push



\# Получить изменения с сервера

git fetch origin



\# Получить изменения в текущую ветку

git pull --rebase



\# Клонировать репозиторий

git clone git@github.com:t-iranazavr/git-docker-course.git



\# Подключить origin, если он еще не задан

git remote add origin git@github.com:t-iranazavr/git-docker-course.git



\# Изменить адрес существующего origin

git remote set-url origin git@github.com:t-iranazavr/git-docker-course.git



\# Отправить ветку и установить upstream

git push -u origin GIT
